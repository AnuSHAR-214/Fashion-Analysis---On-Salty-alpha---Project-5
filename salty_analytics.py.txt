"""
Salty Fashion Analytics — full pipeline
=======================================

One file. Scrapes the catalogue, cleans it, runs the analysis, and writes a
self-contained dashboard with the results baked in.

    pip install requests pandas
    pip install textblob                     # optional, for sentiment
    python -m textblob.download_corpora      # optional

Run it:

    python salty_analytics.py                     # scrape live, then build
    python salty_analytics.py --offline my.csv    # skip scraping, use a CSV
    python salty_analytics.py --limit 5           # scrape only 5 pages (test)

Outputs, written next to this script:

    salty_cleaned.csv                 cleaned data
    salty_insights.json               the analysis, as data
    fashion_analytics_dashboard.html  the dashboard, opens in any browser
"""

import argparse
import json
import sys
import time
from pathlib import Path

import pandas as pd
import requests

BASE = "https://salty.co.in"
COLLECTION = "all-products"     # 1,537 products; use "all" for the whole store
PER_PAGE = 250                  # Shopify's maximum per request
DELAY = 1.0                     # seconds between requests
HEADERS = {"User-Agent": "Mozilla/5.0 (compatible; portfolio-project/1.0)"}
OUT = Path(__file__).resolve().parent


# ======================================================================
# 1. SCRAPE
# ======================================================================
def fetch_products(collection=COLLECTION, max_pages=50):
    """
    salty.co.in runs on Shopify, which serves the catalogue as JSON at
    /products.json. That avoids Selenium, infinite scroll and HTML parsing
    entirely, and it exposes product_type (category), which the rendered
    product cards do not.
    """
    rows, page = [], 1

    while page <= max_pages:
        url = f"{BASE}/collections/{collection}/products.json"
        try:
            resp = requests.get(url, headers=HEADERS, timeout=30,
                                params={"limit": PER_PAGE, "page": page})
        except requests.RequestException as e:
            print(f"  page {page}: request failed ({e}), stopping")
            break

        if resp.status_code != 200:
            print(f"  page {page}: HTTP {resp.status_code}, stopping")
            break

        products = resp.json().get("products", [])
        if not products:
            print(f"  page {page}: empty, done")
            break

        for p in products:
            variant = (p.get("variants") or [{}])[0]
            price = variant.get("price")
            compare = variant.get("compare_at_price")
            rows.append({
                "Product_Name": (p.get("title") or "").strip(),
                "Price":     float(price) if price else None,
                "Old_Price": float(compare) if compare else None,
                "Category":  (p.get("product_type") or "").strip() or None,
                "Tags":      ", ".join(p.get("tags") or []) or None,
                "Available": variant.get("available"),
                "URL":       f"{BASE}/products/{p.get('handle')}",
                "Image":     ((p.get("images") or [{}])[0] or {}).get("src"),
            })

        print(f"  page {page}: {len(products):>3} products  (total {len(rows)})")
        page += 1
        time.sleep(DELAY)

    return pd.DataFrame(rows)

# 2. CLEAN
# ======================================================================
def dedupe_title(s):
    """
    The product cards render each title twice, so HTML scraping yields
    'Red BoxRed Box'. This was corrupting every chart label. products.json
    is clean, but the guard stays in for CSVs scraped the old way.
    """
    s = str(s).strip()
    half = len(s) // 2
    return s[:half] if (len(s) % 2 == 0 and half and s[:half] == s[half:]) else s


def to_number(series):
    """
    Handles 'Rs. 1,299', '₹1,039', '50% OFF', '1299.0'.

    Note the trap: naively deleting every non-digit except '.' turns 'Rs. 1299'
    into '.1299', which parses as 0.1299 and silently destroys every price.
    Drop separators first, then extract the first complete number.
    """
    s = series.astype(str).str.replace(",", "", regex=False)
    return pd.to_numeric(s.str.extract(r"(-?\d+(?:\.\d+)?)", expand=False), errors="coerce")


def clean(df):
    df = df.copy()
    df.columns = [c.strip() for c in df.columns]

    # tolerate common header variations from older CSVs
    renames = {"name": "Product_Name", "title": "Product_Name", "product": "Product_Name",
               "mrp": "Old_Price", "original_price": "Old_Price",
               "selling_price": "Price", "product_category": "Category"}
    df.rename(columns={c: renames[c.lower()] for c in df.columns
                       if c.lower() in renames}, inplace=True)

    if "Product_Name" not in df or "Price" not in df:
        sys.exit("ERROR: need at least Product_Name and Price columns.")

    df["Product_Name"] = df["Product_Name"].map(dedupe_title)
    df = df[df["Product_Name"].str.len() > 0]

    df["Price"] = to_number(df["Price"])
    df = df.dropna(subset=["Price"])
    df = df[df["Price"] > 0]

    # Old_Price was previously cleaned but never cast, so it stayed a string
    # and every savings figure downstream came out wrong.
    if "Old_Price" in df:
        df["Old_Price"] = to_number(df["Old_Price"])
    else:
        df["Old_Price"] = None

    if "Discount" in df:
        stated = to_number(df["Discount"])
        missing = df["Old_Price"].isna() & stated.between(1, 99)
        df.loc[missing, "Old_Price"] = (df.loc[missing, "Price"] /
                                        (1 - stated[missing] / 100)).round()

    df["Old_Price"] = df["Old_Price"].fillna(df["Price"])
    df.loc[df["Old_Price"] < df["Price"], "Old_Price"] = df["Price"]

    # recompute rather than trust the badge — they disagree on real listings
    df["Savings"] = (df["Old_Price"] - df["Price"]).round(2)
    df["Discount"] = ((df["Savings"] / df["Old_Price"]) * 100).round().astype(int)

    if "Category" not in df:
        df["Category"] = None
    df["Category"] = df["Category"].fillna("Uncategorised")

    key = "URL" if "URL" in df else "Product_Name"
    before = len(df)
    df = df.drop_duplicates(subset=[key]).reset_index(drop=True)
    if before != len(df):
        print(f"  removed {before - len(df)} duplicate rows")

    df["Price_Band"] = pd.cut(
        df["Price"], bins=[0, 300, 600, 1000, 1500, float("inf")],
        labels=["Under 300", "300-599", "600-999", "1000-1499", "1500+"], right=False)

    return df


def add_sentiment(df, column="Product_Name"):
    """
    Honest note: TextBlob scores opinion words, and product titles are
    descriptive rather than evaluative, so most score exactly 0.0. The column
    is here because the brief asks for it and because it becomes genuinely
    useful the moment you point it at scraped review text instead of titles.
    """
    try:
        from textblob import TextBlob
    except ImportError:
        print("  textblob not installed, skipping sentiment")
        return df

    df = df.copy()
    df["Polarity"] = df[column].map(lambda t: round(TextBlob(str(t)).sentiment.polarity, 3))
    df["Sentiment"] = pd.cut(df["Polarity"], bins=[-1.001, -0.05, 0.05, 1.0],
                             labels=["Negative", "Neutral", "Positive"]).astype(str)
    flat = (df["Polarity"] == 0).mean()
    if flat > 0.7:
        print(f"  note: {flat:.0%} of titles scored 0.0 polarity — expected for titles")
    return df

def analyse(df):
    """Everything the dashboard displays is computed here, not in the browser."""
    n = len(df)
    price, old, save = df["Price"], df["Old_Price"], df["Savings"]
    total_save, total_old = save.sum(), old.sum()
    mean, med = price.mean(), price.median()
    ins = []

    def add(tone, icon, title, text):
        ins.append({"tone": tone, "icon": icon, "title": title, "text": text})

    rupee = lambda v: f"₹{round(v):,}"

    # spread
    hi, lo = df.loc[price.idxmax()], df.loc[price.idxmin()]
    add("", "i-scales", "Price spread",
        f"The catalogue runs from <em>{rupee(lo.Price)}</em> to <em>{rupee(hi.Price)}</em>, "
        f"a <em>{hi.Price / lo.Price:.1f}×</em> range. The most expensive listing is "
        f"<em>{hi.Product_Name}</em>.")

    # skew
    skew = (mean - med) / med if med else 0
    if abs(skew) > 0.1:
        add("", "i-trend",
            "Pulled up by premium items" if skew > 0 else "Pulled down by cheap items",
            f"Average price is <em>{rupee(mean)}</em> but the median is <em>{rupee(med)}</em>. "
            + ("A small number of expensive listings drag the mean above where most products "
               "actually sit, so quote the median when describing typical pricing."
               if skew > 0 else
               "A cluster of low-priced items sits below the bulk of the range."))
    else:
        add("good", "i-target", "Evenly distributed pricing",
            f"Mean <em>{rupee(mean)}</em> and median <em>{rupee(med)}</em> land within "
            f"<em>{abs(skew) * 100:.0f}%</em> of each other, so there is no strong skew.")

    # discount concentration
    counts = df["Discount"].value_counts()
    top_d, top_c = counts.index[0], counts.iloc[0]
    share = top_c / n
    if share >= 0.3:
        add("warn" if share >= 0.5 else "", "i-alert", "One discount dominates",
            f"<em>{top_c} of {n}</em> products (<em>{share:.0%}</em>) carry exactly "
            f"<em>{top_d}% off</em>. That is a blanket store-wide rule rather than "
            f"per-product pricing, which limits what discount analysis can reveal.")
    else:
        add("", "i-pie", "Varied discounting",
            f"Discounts spread across <em>{counts.size}</em> distinct levels, the most "
            f"common being <em>{top_d}% off</em> on <em>{top_c}</em> products.")

    # pareto
    if total_save > 0:
        ranked = df.sort_values("Savings", ascending=False)
        k = int((ranked["Savings"].cumsum() < total_save / 2).sum()) + 1
        add("warn" if k / n <= 0.3 else "", "i-layers", "Savings are concentrated",
            f"The top <em>{k} products</em> (<em>{k / n:.0%}</em> of the range) account for "
            f"half of all <em>{rupee(total_save)}</em> in savings. "
            f"<em>{ranked.iloc[0].Product_Name}</em> alone contributes "
            f"<em>{rupee(ranked.iloc[0].Savings)}</em>.")

    # blended discount
    blended = total_save / total_old * 100 if total_old else 0
    add("flag" if blended > 45 else "warn" if blended > 30 else "good", "i-tag",
        "Blended discount",
        f"<em>{rupee(total_save)}</em> comes off <em>{rupee(total_old)}</em> of original "
        f"value — an effective <em>{blended:.1f}%</em> discount. "
        + ("Sustained discounting this deep usually means list prices are inflated rather "
           "than genuinely reduced." if blended > 45 else "That is a normal promotional range."))

    # charm pricing
    charm = int((price % 100 == 99).sum())
    if charm / n >= 0.25:
        add("", "i-target", "Charm pricing is the house style",
            f"<em>{charm} of {n}</em> prices end in 99 (<em>{charm / n:.0%}</em>). Pricing is "
            f"set by psychological anchor, so price is a weak proxy for product value here.")

    # outliers
    q1, q3 = price.quantile(.25), price.quantile(.75)
    iqr = q3 - q1
    outs = df[(price > q3 + 1.5 * iqr) | (price < q1 - 1.5 * iqr)]
    if len(outs):
        names = ", ".join(outs.nlargest(3, "Price")["Product_Name"].str.slice(0, 28))
        add("flag", "i-alert", "Price outliers",
            f"<em>{len(outs)}</em> product(s) sit outside the normal range of "
            f"<em>{rupee(q1)}–{rupee(q3)}</em>: <em>{names}</em>"
            f"{' and others' if len(outs) > 3 else ''}. Check these before averaging.")

    # full-price items
    full = int((df["Discount"] == 0).sum())
    if full:
        add("", "i-box", "Undiscounted items",
            f"<em>{full}</em> product(s) (<em>{full / n:.0%}</em>) carry no discount at all. "
            f"They flatten the average savings figure and are worth excluding from "
            f"discount-depth analysis.")

    # categories
    cats = df[df["Category"] != "Uncategorised"]
    if cats["Category"].nunique() > 1:
        by_count = cats["Category"].value_counts()
        by_value = cats.groupby("Category")["Price"].sum().sort_values(ascending=False)
        add("", "i-grid", "Category mix",
            f"<em>{by_count.size}</em> categories present. <em>{by_count.index[0]}</em> leads "
            f"on count with <em>{by_count.iloc[0]} products</em> "
            f"(<em>{by_count.iloc[0] / len(cats):.0%}</em>), while <em>{by_value.index[0]}</em> "
            f"carries the most catalogue value at <em>{rupee(by_value.iloc[0])}</em>.")

        deep = cats.groupby("Category")["Discount"].mean().sort_values(ascending=False)
        add("", "i-bars", "Discount depth by category",
            f"<em>{deep.index[0]}</em> is discounted hardest at <em>{deep.iloc[0]:.0f}%</em> "
            f"on average, against <em>{deep.iloc[-1]:.0f}%</em> for <em>{deep.index[-1]}</em>.")

    # sentiment
    if "Sentiment" in df and df["Sentiment"].notna().any():
        s = df["Sentiment"].value_counts()
        pos, neu, neg = int(s.get("Positive", 0)), int(s.get("Neutral", 0)), int(s.get("Negative", 0))
        add("good" if pos > neg * 2 else "flag" if neg > pos else "", "i-spark",
            "Title sentiment",
            f"<em>{pos} positive</em>, <em>{neu} neutral</em>, <em>{neg} negative</em> across "
            f"<em>{n}</em> product titles. Titles are descriptive rather than evaluative, so "
            f"treat this as a text-processing demonstration, not a customer signal.")

    # best value
    ratio = (save / price).replace([float("inf")], 0)
    if ratio.max() > 0:
        best = df.loc[ratio.idxmax()]
        add("good", "i-coins", "Deepest value",
            f"<em>{best.Product_Name}</em> returns <em>₹{best.Savings / best.Price:.2f}</em> "
            f"saved per rupee spent — the strongest discount-to-price ratio in the catalogue.")

    kpis = {
        "products": int(n),
        "average_price": round(float(mean), 2),
        "median_price": round(float(med), 2),
        "highest_price": round(float(price.max()), 2),
        "highest_name": str(df.loc[price.idxmax(), "Product_Name"]),
        "lowest_price": round(float(price.min()), 2),
        "lowest_name": str(df.loc[price.idxmin(), "Product_Name"]),
        "total_savings": round(float(total_save), 2),
        "blended_discount": round(float(blended), 1),
        "categories": int(cats["Category"].nunique()),
    }
    return kpis, ins

# 4. TREND ANALYSIS
# ======================================================================
STOPWORDS = {
    "the", "and", "for", "with", "her", "his", "him", "you", "your", "our",
    "set", "of", "in", "on", "to", "a", "an", "by", "from", "free", "new",
    "salty", "gift", "pack", "combo", "style", "design", "women", "men",
    "womens", "mens", "girls", "boys", "piece", "pcs", "size", "color",
    "colour", "make", "it", "is", "are", "best", "top",
}

COLOURS = ["rose gold", "gold", "silver", "black", "white", "pink", "purple",
           "blue", "green", "red", "multi", "yellow", "brown", "grey"]

MATERIALS = ["pearl", "crystal", "stainless steel", "copper", "alloy", "bead",
             "stone", "zircon", "resin", "leather", "thread", "acrylic", "enamel"]


def trend_analysis(df, top_n=12):
    """
    What's actually trending, using signals the catalogue really carries.

    The strongest one is sell-out rate: an out-of-stock product sold through,
    so the share of unavailable items in a theme is a demand proxy rather than
    a guess. Launch recency is the second — what the brand chose to add most
    recently is what it is betting on now.
    """
    import re
    from collections import Counter

    n = len(df)
    out = {"motifs": [], "colours": [], "materials": [], "launches": [],
           "has_stock": False, "has_dates": False}

    titles = df["Product_Name"].astype(str).str.lower()
    has_stock = "Available" in df and df["Available"].notna().any()
    out["has_stock"] = bool(has_stock)

    # ---- motifs: what words recur across product names ----
    tokens = Counter()
    for t in titles:
        for w in set(re.findall(r"[a-z]+", t)):
            if len(w) > 2 and w not in STOPWORDS:
                tokens[w] += 1

    for word, count in tokens.most_common(top_n):
        if count < max(2, n * 0.01):
            continue
        sub = df[titles.str.contains(rf"\b{re.escape(word)}\b", regex=True)]
        row = {
            "term": word.title(),
            "count": int(len(sub)),
            "share": round(len(sub) / n * 100, 1),
            "avg_price": round(float(sub["Price"].mean())),
            "avg_discount": round(float(sub["Discount"].mean())),
        }
        if has_stock:
            row["sold_out"] = round(float((~sub["Available"].astype(bool)).mean()) * 100, 1)
        out["motifs"].append(row)

    # ---- colour and material vocabulary ----
    haystack = titles
    if "Tags" in df:
        haystack = titles + " " + df["Tags"].fillna("").astype(str).str.lower()

    for label, vocab in (("colours", COLOURS), ("materials", MATERIALS)):
        found = []
        for term in vocab:
            mask = haystack.str.contains(rf"\b{re.escape(term)}", regex=True)
            hits = int(mask.sum())
            if hits >= max(2, n * 0.005):
                sub = df[mask]
                entry = {"term": term.title(), "count": hits,
                         "share": round(hits / n * 100, 1),
                         "avg_price": round(float(sub["Price"].mean()))}
                if has_stock:
                    entry["sold_out"] = round(float((~sub["Available"].astype(bool)).mean()) * 100, 1)
                found.append(entry)
        # 'gold' would double-count 'rose gold', so drop the overlap
        if label == "colours":
            rose = next((f for f in found if f["term"] == "Rose Gold"), None)
            gold = next((f for f in found if f["term"] == "Gold"), None)
            if rose and gold:
                gold["count"] = max(0, gold["count"] - rose["count"])
                gold["share"] = round(gold["count"] / n * 100, 1)
        out[label] = sorted(found, key=lambda x: -x["count"])[:top_n]

    # ---- launch timeline ----
    if "Published" in df and df["Published"].notna().any():
        dates = pd.to_datetime(df["Published"], errors="coerce", utc=True)
        if dates.notna().any():
            out["has_dates"] = True
            monthly = dates.dt.to_period("M").value_counts().sort_index().tail(12)
            out["launches"] = [{"month": str(p), "count": int(c)}
                               for p, c in monthly.items()]
            recent = dates >= (dates.max() - pd.Timedelta(days=90))
            out["recent_share"] = round(float(recent.mean()) * 100, 1)
            if "Category" in df:
                rc = df[recent.fillna(False)]["Category"].value_counts().head(5)
                out["recent_categories"] = [{"name": str(k), "count": int(v)}
                                            for k, v in rc.items()]
    return out


def trend_insights(df, tr):
    """Turn the trend numbers into the same insight format as the rest."""
    ins = []
    add = lambda tone, icon, title, text: ins.append(
        {"tone": tone, "icon": icon, "title": title, "text": text})

    if tr["motifs"]:
        m = tr["motifs"][0]
        add("", "i-spark", "Dominant motif",
            f"<em>{m['term']}</em> appears in <em>{m['count']} products</em> "
            f"(<em>{m['share']}%</em> of the catalogue), averaging "
            f"<em>₹{m['avg_price']:,}</em> at <em>{m['avg_discount']}%</em> off. "
            f"It is the single most repeated theme in the range.")

    if tr["has_stock"] and tr["motifs"]:
        ranked = [m for m in tr["motifs"] if m.get("sold_out") is not None]
        if ranked:
            hot = max(ranked, key=lambda x: x["sold_out"])
            cold = min(ranked, key=lambda x: x["sold_out"])
            add("good", "i-trend", "Selling through fastest",
                f"<em>{hot['sold_out']}%</em> of <em>{hot['term']}</em> products are out of "
                f"stock, against <em>{cold['sold_out']}%</em> for <em>{cold['term']}</em>. "
                f"Sell-through is a demand signal the price columns cannot show.")

    if tr["colours"]:
        c = tr["colours"][0]
        add("", "i-pie", "Leading colour",
            f"<em>{c['term']}</em> leads on colour with <em>{c['count']} products</em> "
            f"(<em>{c['share']}%</em>), averaging <em>₹{c['avg_price']:,}</em>.")

    if tr["materials"]:
        mat = tr["materials"][0]
        add("", "i-layers", "Leading material",
            f"<em>{mat['term']}</em> is the most common material at "
            f"<em>{mat['count']} products</em> (<em>{mat['share']}%</em>).")

    if tr.get("has_dates") and tr.get("recent_share") is not None:
        cats = tr.get("recent_categories") or []
        lead = f" Recent launches skew towards <em>{cats[0]['name']}</em>." if cats else ""
        add("warn" if tr["recent_share"] > 40 else "", "i-bars", "Launch velocity",
            f"<em>{tr['recent_share']}%</em> of the catalogue was published in the last "
            f"90 days.{lead} A high share means the range turns over fast, so any "
            f"snapshot analysis dates quickly.")
    return ins

def plot_charts(df, tr, folder="charts"):
    """Static PNGs for slides, the report, or importing into Power BI."""
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt
    except ImportError:
        print("  matplotlib not installed, skipping charts")
        return []

    # Lavender only. The mint and rose from the dashboard palette are semantic
    # (good / bad), so using them as categorical colours misreads as a signal.
    LAV = ["#5B3FA8", "#8B5FD6", "#B07BE8", "#6E7FE0", "#C98FD9",
           "#3F2B78", "#9A8AF0", "#D9A8E6", "#8E6FC4", "#7C63D6"]
    INK, GRID, PAPER = "#2E2350", "#E6DCF8", "#F6F1FD"

    d = folder if isinstance(folder, Path) else OUT / folder
    d.mkdir(exist_ok=True)
    made = []

    def style(ax, title):
        ax.set_title(title, color=INK, fontsize=13, weight="bold", pad=14, loc="left")
        ax.set_facecolor(PAPER)
        ax.tick_params(colors=INK, labelsize=9)
        for s in ("top", "right"):
            ax.spines[s].set_visible(False)
        for s in ("left", "bottom"):
            ax.spines[s].set_color(GRID)
        ax.grid(axis="x", color=GRID, lw=1)
        ax.set_axisbelow(True)

    def save(fig, name):
        path = d / name
        fig.patch.set_facecolor(PAPER)
        fig.tight_layout()
        fig.savefig(path, dpi=150, facecolor=PAPER)
        plt.close(fig)
        made.append(path)

    # price distribution
    fig, ax = plt.subplots(figsize=(8, 4.2))
    ax.hist(df["Price"], bins=min(30, max(6, len(df) // 8)),
            color=LAV[1], edgecolor=PAPER)
    ax.axvline(df["Price"].median(), color=LAV[0], ls="--", lw=2,
               label=f"median ₹{df['Price'].median():,.0f}")
    ax.legend(frameon=False, labelcolor=INK, fontsize=9)
    style(ax, "Price distribution")
    ax.grid(axis="x", visible=False)
    ax.grid(axis="y", color=GRID, lw=1)
    save(fig, "01_price_distribution.png")

    # categories
    cats = df[df["Category"] != "Uncategorised"]["Category"].value_counts().head(10)
    if len(cats) > 1:
        fig, ax = plt.subplots(figsize=(8, 4.2))
        ax.barh(cats.index[::-1], cats.values[::-1],
                color=[LAV[i % len(LAV)] for i in range(len(cats))])
        style(ax, "Products by category")
        save(fig, "02_categories.png")

        avg = df[df["Category"] != "Uncategorised"].groupby("Category")["Discount"].mean()
        avg = avg.sort_values().tail(10)
        fig, ax = plt.subplots(figsize=(8, 4.2))
        ax.barh(avg.index, avg.values, color=LAV[2])
        ax.set_xlabel("average discount %", color=INK, fontsize=9)

def build_dashboard(df, kpis, insights, source, trends=None):
    cols = ["Product_Name", "Price", "Old_Price", "Discount", "Savings", "Category"]
    if "Sentiment" in df:
        cols.append("Sentiment")

    rows = df[cols].rename(columns={
        "Product_Name": "n", "Price": "p", "Old_Price": "o",
        "Discount": "d", "Savings": "s", "Category": "c", "Sentiment": "sent",
    })
    rows["p"] = rows["p"].round(0).astype(int)
    rows["o"] = rows["o"].round(0).astype(int)
    rows["s"] = rows["s"].round(0).astype(int)

    payload = json.dumps({
        "rows": rows.to_dict("records"),
        "kpis": kpis,
        "insights": insights,
        "trends": trends or {},
        "source": source,
    }, ensure_ascii=False, separators=(",", ":"))

    html = TEMPLATE.replace("/*__PAYLOAD__*/null", payload)
    path = OUT / "fashion_analytics_dashboard.html"
    path.write_text(html, encoding="utf-8")
    return path

function renderDonut(rows){
  const host=document.getElementById("donut"),keys=document.getElementById("donutKeys");

  host.innerHTML="";keys.innerHTML="";
  if(!rows.length){keys.innerHTML='<p class="empty">No data.</p>';return}
  const distinct=new Set(rows.map(r=>r.d));let items;
  if(distinct.size>8){
    const defs=[["No discount",r=>r.d===0],["Under 20%",r=>r.d>0&&r.d<20],["20–29%",r=>r.d>=20&&r.d<30],
      ["30–39%",r=>r.d>=30&&r.d<40],["40–49%",r=>r.d>=40&&r.d<50],["Exactly 50%",r=>r.d===50],
      ["51–69%",r=>r.d>50&&r.d<70],["70% and over",r=>r.d>=70]];
    items=defs.map(([l,f])=>({label:l,c:rows.filter(f).length})).filter(i=>i.c);
    note.textContent=`Grouped into bands — ${distinct.size} distinct discount values present.`;
  }else{
    const t={};rows.forEach(r=>t[r.d]=(t[r.d]||0)+1);
    items=Object.entries(t).map(([d,c])=>({label:d+"% off",c,d:+d}))
      .sort((a,b)=>b.c-a.c||b.d-a.d);
    note.textContent="Products grouped by the discount on the listing.";
  }
  const R=62,TH=20,C=2*Math.PI*R,S=160;let off=0;
  const svg=[`<svg width="${S}" height="${S}" viewBox="0 0 ${S} ${S}" role="img" aria-label="Discount distribution">`];
  items.forEach((it,i)=>{const f=it.c/rows.length;
    svg.push(`<circle cx="${S/2}" cy="${S/2}" r="${R}" fill="none" stroke="${RAMP[i%RAMP.length]}"
      stroke-width="${TH}" stroke-dasharray="${(f*C).toFixed(2)} ${C.toFixed(2)}"
      stroke-dashoffset="${(-off*C).toFixed(2)}" transform="rotate(-90 ${S/2} ${S/2})">
      <title>${esc(it.label)} — ${it.c} products</title></circle>`);

function squarify(items,x,y,w,h,out){
  if(!items.length)return out;
  if(items.length===1){out.push({...items[0],x,y,w,h});return out}
  const total=items.reduce((a,b)=>a+b.v,0);let split=0,acc=0;
  for(let i=0;i<items.length;i++){acc+=items[i].v;split=i+1;if(acc>=total/2)break}
  const f=acc/total;
  if(w>=h){squarify(items.slice(0,split),x,y,w*f,h,out);
           squarify(items.slice(split),x+w*f,y,w*(1-f),h,out)}
  else{squarify(items.slice(0,split),x,y,w,h*f,out);
       squarify(items.slice(split),x,y+h*f,w,h*(1-f),out)}
  return out;
}

def main():
    ap = argparse.ArgumentParser(description="Scrape, clean, analyse and build the dashboard.")
    ap.add_argument("--offline", metavar="CSV", help="use an existing CSV instead of scraping")
    ap.add_argument("--limit", type=int, default=50, help="max pages to scrape")
    ap.add_argument("--collection", default=COLLECTION)
    ap.add_argument("--no-sentiment", action="store_true")
    ap.add_argument("--no-charts", action="store_true", help="skip the PNG charts")
    args = ap.parse_args()

    if args.offline:
        print(f"Reading {args.offline}")
        raw = pd.read_csv(args.offline)
        source = Path(args.offline).stem
    else:
        print(f"Scraping /collections/{args.collection}/products.json")
        raw = fetch_products(args.collection, args.limit)
        source = f"salty.co.in/{args.collection}"

    if raw.empty:
        sys.exit("No products fetched. Check the collection name or your connection.")

    print(f"\nCleaning {len(raw)} rows")
    df = clean(raw)
    if not args.no_sentiment:
        df = add_sentiment(df)
    print(f"  {len(df)} rows kept")

    df.to_csv(OUT / "salty_cleaned.csv", index=False, encoding="utf-8-sig")

    kpis, insights = analyse(df)

    tr = trend_analysis(df)
    insights = insights + trend_insights(df, tr)

    (OUT / "salty_insights.json").write_text(
        json.dumps({"kpis": kpis, "insights": insights, "trends": tr},
                   indent=2, ensure_ascii=False, default=str),
        encoding="utf-8")

    charts = [] if args.no_charts else plot_charts(df, tr)
    path = build_dashboard(df, kpis, insights, source, tr)

    print("\n" + "=" * 62)
    print(f"  Products        {kpis['products']:,}")
    print(f"  Categories      {kpis['categories']}")
    print(f"  Average price   Rs.{kpis['average_price']:,.0f}   (median Rs.{kpis['median_price']:,.0f})")
    print(f"  Price range     Rs.{kpis['lowest_price']:,.0f} – Rs.{kpis['highest_price']:,.0f}")
    print(f"  Total savings   Rs.{kpis['total_savings']:,.0f}")
    print(f"  Blended discount {kpis['blended_discount']}%")
    print("=" * 62)

    if tr["motifs"]:
        print("\n  Top motifs")
        for m in tr["motifs"][:6]:
            sold = f"  {m['sold_out']:>5}% sold out" if "sold_out" in m else ""
            print(f"    {m['term']:<14} {m['count']:>4} products  "
                  f"Rs.{m['avg_price']:>6,}  {m['avg_discount']:>3}% off{sold}")

    if tr["colours"]:
        print("\n  Colours: " + ", ".join(f"{c['term']} ({c['count']})" for c in tr["colours"][:6]))
    if tr["materials"]:
        print("  Materials: " + ", ".join(f"{m['term']} ({m['count']})" for m in tr["materials"][:6]))
    if tr.get("recent_share") is not None:
        print(f"\n  {tr['recent_share']}% of the catalogue was published in the last 90 days")

    print(f"\n  {len(insights)} findings:")
    for i in insights:
        print(f"    • {i['title']}")

    print(f"\n  salty_cleaned.csv")
    print(f"  salty_insights.json")
    print(f"  {path.name}")
    for c in charts:
        print(f"  charts/{c.name}")


if __name__ == "__main__":
    main()
